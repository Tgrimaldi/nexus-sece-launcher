name: Deploy NEXUS SECE Launcher
on:
  push:
    branches: [main]
  workflow_dispatch:
permissions:
  contents: read
  pages: write
  id-token: write
concurrency:
  group: pages
  cancel-in-progress: true
jobs:
  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Verify PWA package
        run: |
          test -f index.html
          test -f icon-180.png
          test -f icon-192.png
          test -f icon-512.png
          test -f icon-maskable-512.png
          test -f banyoles-operari/manifest.webmanifest
          test -f banyoles-toni/manifest.webmanifest
          test -f lloret-operari/manifest.webmanifest
          test -f lloret-jaume/manifest.webmanifest
      - uses: actions/configure-pages@v5
        with:
          enablement: true
      - uses: actions/upload-pages-artifact@v3
        with:
          path: .
      - name: Deploy
        id: deployment
        uses: actions/deploy-pages@v4
